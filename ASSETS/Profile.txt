const wxColour kAvatarSlate(104, 116, 138);
const wxColour kAvatarLight(236, 240, 246);

// ---- big default avatar with camera badge (custom drawn) --------------------
class AvatarCircle : public wxPanel {
public:
    explicit AvatarCircle(wxWindow* parent) : wxPanel(parent, wxID_ANY) {
        SetBackgroundStyle(wxBG_STYLE_PAINT);
        SetMinSize(FromDIP(wxSize(124, 124)));
        Bind(wxEVT_PAINT, [this](wxPaintEvent&) { OnPaint(); });
    }
private:
    void OnPaint() {
        wxAutoBufferedPaintDC dc(this);
        dc.SetBackground(wxBrush(GetParent()->GetBackgroundColour()));
        dc.Clear();
        const wxSize s = GetClientSize();
        const wxPoint c(s.x / 2, s.y / 2);
        const double R = std::min(s.x, s.y) / 2.0 - 4;

        dc.SetPen(*wxTRANSPARENT_PEN);
        dc.SetBrush(wxBrush(kAvatarSlate));
        dc.DrawCircle(c, int(R));

        // silhouette, clipped to the inner disc
        std::vector<wxPoint> disc;
        for (int i = 0; i < 72; ++i) {
            const double a = i * 3.14159265358979 / 36.0;
            disc.push_back(wxPoint(c.x + int(R * 0.86 * std::cos(a)), c.y + int(R * 0.86 * std::sin(a))));
        }
        dc.SetClippingRegion(wxRegion(disc.size(), disc.data()));
        dc.SetBrush(wxBrush(kAvatarLight));
        dc.DrawCircle(c, int(R * 0.86));
        dc.SetBrush(wxBrush(kAvatarSlate));
        dc.DrawCircle(c.x, c.y - int(R * 0.18), int(R * 0.30));                       // head
        dc.DrawEllipse(c.x - int(R * 0.66), c.y + int(R * 0.30), int(R * 1.32), int(R * 1.0));  // shoulders
        dc.DestroyClippingRegion();

        // camera badge
        const wxPoint b(c.x + int(R * 0.72), c.y + int(R * 0.72));
        dc.SetPen(wxPen(*wxWHITE, 2));
        dc.SetBrush(wxBrush(kLpRed));
        dc.DrawCircle(b, FromDIP(13));
        const wxBitmap cam = SvgBundle(kSvgCamera, *wxWHITE, 13).GetBitmapFor(this);
        dc.DrawBitmap(cam, b.x - cam.GetWidth() / 2, b.y - cam.GetHeight() / 2, true);
    }
};
